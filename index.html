TOKENIZATION – WORD AND SENTENCE TOKENIZATION

Splits text into constituent sentences using regular expressions and breaks sentences into word tokens using NLTK's wordpunct_tokenize without requiring external downloads.

Python (Offline) 
import nltk
import re
from nltk.tokenize import wordpunct_tokenize


def sentence_tokenize(text):
    sentences = re.split(r'[.!?]+', text)

    result = []

    for sentence in sentences:
        sentence = sentence.strip()

        if sentence:
            result.append(sentence)

    return result


def tokenize_text(text):
    sentences = sentence_tokenize(text)

    print("Sentence Tokens:")

    for i in range(len(sentences)):
        print(i + 1, ":", sentences[i])

    print("\nWord Tokens:")

    for i in range(len(sentences)):
        words = wordpunct_tokenize(sentences[i])

        print("Sentence", i + 1, ":", words)


def main():
    text = "Natural Language Processing is interesting. It is used in many applications. NLP helps computers understand human language."

    tokenize_text(text)


main()

STOPWORD ELIMINATION

Filters high-frequency common words (stopwords) from input text using a custom predefined offline set, displaying token lists and reduction counts.

Python (Offline) 
import nltk
from nltk.tokenize import wordpunct_tokenize


STOPWORDS = {
    "a", "an", "the", "is", "are", "was", "were",
    "am", "and", "or", "but", "in", "on", "at",
    "to", "of", "for", "with", "this", "that",
    "it", "as", "by", "from", "be", "been",
    "has", "have", "had"
}


def remove_stopwords(text):
    words = wordpunct_tokenize(text)

    filtered_words = []

    for word in words:
        if word.lower() not in STOPWORDS:
            filtered_words.append(word)

    print("Original Words:")
    print(words)

    print("\nAfter Stopword Removal:")
    print(filtered_words)

    print("\nOriginal Count:", len(words))
    print("After Removal Count:", len(filtered_words))


def main():
    text = "This is a simple example of natural language processing and text analysis."

    remove_stopwords(text)


main()
Expected Output

After Stopword Removal:

Original Count: 13

STEMMING

Reduces inflected words to their root/stem forms using PorterStemmer algorithm from NLTK, showcasing both full sentence and word-level stemming.

Python (Offline) 
import nltk
from nltk.tokenize import wordpunct_tokenize
from nltk.stem import PorterStemmer


def stemming(text):
    stemmer = PorterStemmer()

    words = wordpunct_tokenize(text)

    stemmed_words = []

    for word in words:
        stemmed_words.append(stemmer.stem(word))

    print("Original Text:")
    print(text)

    print("\nStemmed Text:")
    print(" ".join(stemmed_words))

    print("\nStemming Check:")
    print("building ->", stemmer.stem("building"))
    print("programs ->", stemmer.stem("programs"))


def main():
    text = "The students are building programs and studying NLP."

    stemming(text)


main()

Stemmed Text:

PART OF SPEECH TAGGING – VERSION 1 (STANDARD)

Assigns syntactic grammatical categories (nouns, verbs, determiners, pronouns, prepositions, adjectives) using NLTK's RegexpTagger with comprehensive rules offline.

Python (Offline) 
import nltk
from nltk.tag import RegexpTagger
from nltk.tokenize import wordpunct_tokenize


def pos_tagging(sentence):

    patterns = [
        (r'^(a|an|the|this|that)$', 'DT'),
        (r'^(i|you|he|she|we|they|it)$', 'PRP'),
        (r'^(is|am|are|was|were|be|been|being)$', 'VB'),
        (r'.*ing$', 'VBG'),
        (r'.*ed$', 'VBD'),
        (r'^(in|on|at|by|with|from|to|for)$', 'IN'),
        (r'^(and|or|but)$', 'CC'),
        (r'.*ly$', 'RB'),
        (r'.*(ful|ous|ive|al)$', 'JJ'),
        (r'^[.,!?]$', '.'),
        (r'.*$', 'NN')
    ]

    tagger = RegexpTagger(patterns)

    words = wordpunct_tokenize(sentence)

    lower_words = []

    for word in words:
        lower_words.append(word.lower())

    tagged_words = tagger.tag(lower_words)

    print("POS Tags:")

    for i in range(len(words)):
        print(words[i], "->", tagged_words[i][1])


def main():
    sentence = "The students are studying natural language processing."

    pos_tagging(sentence)


main()

studying -> VBG
natural -> JJ
language -> NN
processing -> VBG
. -> .

LEMMATIZATION

Resolves words to meaningful canonical dictionary lemmas (e.g., mice → mouse, children → child) using custom rule mappings without requiring WordNet downloads.

Python (Offline) 
import nltk
from nltk.tokenize import wordpunct_tokenize


LEMMA_RULES = {
    "cats": "cat",
    "dogs": "dog",
    "mice": "mouse",
    "children": "child",
    "men": "man",
    "women": "woman",
    "programs": "program",
    "students": "student"
}


def lemmatize_text(text):

    words = wordpunct_tokenize(text)

    lemmatized_words = []

    for word in words:
        lower_word = word.lower()

        if lower_word in LEMMA_RULES:
            lemmatized_words.append(LEMMA_RULES[lower_word])
        else:
            lemmatized_words.append(word)

    print("Original Words:")
    print(words)

    print("\nLemmatized Words:")
    print(lemmatized_words)


def main():
    text = "The cats are chasing mice."

    lemmatize_text(text)


main()
Expected Output 
Original Words:
['The', 'cats', 'are', 'chasing', 'mice', '.']

Lemmatized Words:
['The', 'cat', 'are', 'chasing', 'mouse', '.']

CHUNKING – VERSION 1 (WITH DYNAMIC TAGGER)

Performs full shallow parsing by combining RegexpTagger and RegexpParser to parse sentence tokens into structured constituent phrases (NP, PP, VP) and print the parse tree.

Python (Offline) 
import nltk
from nltk.tag import RegexpTagger
from nltk.tokenize import wordpunct_tokenize
from nltk.chunk import RegexpParser


def chunking(sentence):

    patterns = [
        (r'^(a|an|the|this|that)$', 'DT'),
        (r'.*(ful|ous|ive|al)$', 'JJ'),
        (r'^(is|am|are|was|were|be|been)$', 'VB'),
        (r'.*ing$', 'VBG'),
        (r'.*ed$', 'VBD'),
        (r'^(in|on|at|by|with|from|to|for)$', 'IN'),
        (r'^[.,!?]$', '.'),
        (r'.*$', 'NN')
    ]

    tagger = RegexpTagger(patterns)

    words = wordpunct_tokenize(sentence)

    lower_words = []

    for word in words:
        lower_words.append(word.lower())

    tagged = tagger.tag(lower_words)

    grammar = r"""
        NP: {
*}
        PP: {}
        VP: {*}
    """

    parser = RegexpParser(grammar)

    tree = parser.parse(tagged)

    print(tree)


def main():
    sentence = "The beautiful dog is playing in the park."

    chunking(sentence)


main()
Expected Output 
(S
  (NP the/DT beautiful/JJ dog/NN)
  (VP is/VB (PP in/IN (NP the/DT park/NN)))
  playing/VBG
  ./.)

NAMED ENTITY RECOGNITION

Identifies proper noun entities such as PERSON, ORGANIZATION, and GPE (Geopolitical Entity/Location) using offline dictionary lookup sets.

Python (Offline) 
import nltk
from nltk.tokenize import wordpunct_tokenize


PERSONS = {
    "john",
    "rahul",
    "rohit",
    "aishwarya"
}

ORGANIZATIONS = {
    "google",
    "microsoft",
    "amazon",
    "ibm"
}

PLACES = {
    "india",
    "bangalore",
    "bengaluru",
    "delhi",
    "london"
}


def named_entity_recognition(sentence):

    words = wordpunct_tokenize(sentence)

    for word in words:

        lower_word = word.lower()

        if lower_word in PERSONS:
            print(word, "-> PERSON")

        elif lower_word in ORGANIZATIONS:
            print(word, "-> ORGANIZATION")

        elif lower_word in PLACES:
            print(word, "-> GPE")

        else:
            print(word, "-> O")


def main():
    sentence = "Rahul works at Google in Bangalore."

    named_entity_recognition(sentence)


main()

Google -> ORGANIZATION
in -> O

. -> O

TF-IDF

Computes Term Frequency – Inverse Document Frequency values for document representations using math.log and custom text preprocessing.

Python (Offline) 
import nltk
import math
import string
from nltk.tokenize import wordpunct_tokenize


STOPWORDS = {
    "a", "an", "the", "is", "are",
    "and", "or", "of", "to", "in",
    "on", "for", "with", "this",
    "that", "it"
}


def preprocess_text(text):

    tokens = wordpunct_tokenize(text)

    words = []

    for token in tokens:

        token = token.lower()

        if token not in string.punctuation:
            if token not in STOPWORDS:
                words.append(token)

    return words


def calculate_tf(words, term):

    count = 0

    for word in words:
        if word == term:
            count += 1

    if len(words) == 0:
        return 0

    return count / len(words)


def calculate_idf(documents, term):

    count = 0

    for document in documents:

        if term in document:
            count += 1

    return math.log(len(documents) / count)


def tf_idf(documents):

    vocabulary = set()

    for document in documents:
        for word in document:
            vocabulary.add(word)

    vocabulary = sorted(vocabulary)

    print("Vocabulary:")
    print(vocabulary)

    print("\nTF-IDF Values:")

    for i in range(len(documents)):

        print("\nDocument", i + 1)

        for term in vocabulary:

            tf = calculate_tf(documents[i], term)
            idf = calculate_idf(documents, term)

            value = tf * idf

            if value != 0:
                print(term, ":", round(value, 4))


def main():

    texts = [
        "Natural language processing is interesting",
        "Natural language processing is useful",
        "Machine learning is interesting"
    ]

    documents = []

    for text in texts:
        documents.append(preprocess_text(text))

    tf_idf(documents)


main()

CYK PARSING

Implements Cocke-Younger-Kasami (CYK) bottom-up dynamic programming chart parsing algorithm to check if a sentence is grammatically valid against Chomsky Normal Form (CNF) rules.

Python (Offline) 
import nltk


def cyk_parser(words, grammar):

    n = len(words)

    table = [[set() for j in range(n)] for i in range(n)]

    for i in range(n):

        word = words[i]

        for lhs in grammar:

            for rhs in grammar[lhs]:

                if len(rhs) == 1 and rhs[0] == word:
                    table[i][i].add(lhs)

    for length in range(2, n + 1):

        for i in range(n - length + 1):

            j = i + length - 1

            for k in range(i, j):

                left = table[i][k]
                right = table[k + 1][j]

                for lhs in grammar:

                    for rhs in grammar[lhs]:

                        if len(rhs) == 2:

                            if rhs[0] in left and rhs[1] in right:
                                table[i][j].add(lhs)

    print("CYK Table:")

    for row in table:
        print(row)

    if "S" in table[0][n - 1]:
        print("\nSentence is VALID according to the grammar.")
    else:
        print("\nSentence is NOT VALID according to the grammar.")


def main():

    grammar = {

        "S": [
            ("NP", "VP")
        ],

        "NP": [
            ("John",)
        ],

        "VP": [
            ("likes", "NP")
        ]
    }

    words = ["John", "likes", "John"]

    cyk_parser(words, grammar)


main()

UNIGRAM, BIGRAM AND TRIGRAM

Generates continuous contiguous sequences of 1, 2, and 3 tokens (Unigrams, Bigrams, Trigrams) using NLTK's ngrams utility function.

Python (Offline) 
import nltk
from nltk.tokenize import wordpunct_tokenize
from nltk.util import ngrams


def generate_ngrams(text):

    tokens = wordpunct_tokenize(text)

    for n in range(1, 4):

        print("\n", n, "-gram:")

        result = ngrams(tokens, n)

        for item in result:
            print(item)


def main():

5B

BIGRAM PROBABILITY

Calculates the conditional probability of a given sentence using Bigram language modeling and frequency counts from training sentences.

Python (Offline) 
import nltk
from nltk.tokenize import wordpunct_tokenize


def read_data(sentences):

    words = []

    for sentence in sentences:

        tokens = wordpunct_tokenize(sentence)

        for token in tokens:
            words.append(token.lower())

    return words


def create_counts(words):

    unigram_counts = {}
    bigram_counts = {}

    for word in words:

        if word not in unigram_counts:
            unigram_counts[word] = 0

        unigram_counts[word] += 1

    for i in range(len(words) - 1):

        pair = (words[i], words[i + 1])

        if pair not in bigram_counts:
            bigram_counts[pair] = 0

        bigram_counts[pair] += 1

    return unigram_counts, bigram_counts


def calculate_probability(sentence, unigram_counts, bigram_counts):

    words = wordpunct_tokenize(sentence)

    words = [word.lower() for word in words]

    probability = 1.0

    for i in range(len(words) - 1):

        first = words[i]
        second = words[i + 1]

        bigram = (first, second)

        if first not in unigram_counts:
            return 0

        if bigram not in bigram_counts:
            return 0

        value = bigram_counts[bigram] / unigram_counts[first]

        probability = probability * value

    return probability


def main():

    sentences = [
        "this is my cat",
        "this is my dog",
        "this is your cat",
        "my cat is cute"
    ]

    words = read_data(sentences)

    unigram_counts, bigram_counts = create_counts(words)

    test_sentence = "this is my cat"

    probability = calculate_probability(
        test_sentence,
        unigram_counts,
        bigram_counts
    )

    print("Sentence:", test_sentence)

    print("Bigram Probability:", probability)


main()
Expected Output
